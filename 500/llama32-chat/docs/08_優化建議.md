```python
# cache_manager.py
from functools import lru_cache
import hashlib
import json

class ResponseCache:
    def __init__(self, max_size=1000, ttl=3600):
        self.cache = {}
        self.max_size = max_size
        self.ttl = ttl

    def get_cache_key(self, query, model):
        """生成緩存鍵"""
        content = f"{query}_{model}"
        return hashlib.md5(content.encode()).hexdigest()

    def get(self, query, model):
        """獲取緩存"""
        key = self.get_cache_key(query, model)
        if key in self.cache:
            entry = self.cache[key]
            if time.time() - entry['timestamp'] < self.ttl:
                return entry['response']
        return None

    def set(self, query, model, response):
        """設置緩存"""
        key = self.get_cache_key(query, model)
        self.cache[key] = {
            'response': response,
            'timestamp': time.time()
        }

        # 清理過期緩存
        if len(self.cache) > self.max_size:
            self._cleanup()

# 使用
cache = ResponseCache()
response = cache.get(query, model)
if response is None:
    response = agent.chat(query, model=model)
    cache.set(query, model, response)
```

**預期改善**: 相似查詢響應時間從 245ms 降至 <10ms（95% 命中率下）

class ConversationManager:
    def __init__(self, page_size=100):
        self.page_size = page_size
        self.current_page = 0

    def load_page(self, page_num):
        """按頁加載對話"""
        start = page_num * self.page_size
        end = start + self.page_size
        return self.conversations[start:end]

    def lazy_load(self):
        """延遲加載對話"""
        for i in range(0, len(self.conversations), self.page_size):
            yield self.conversations[i:i+self.page_size]

```python
import functools
import logging

def handle_errors(func):
    """全局錯誤處理裝飾器"""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except requests.exceptions.Timeout:
            logging.error(f"{func.__name__}: 請求超時")
            return {"error": "請求超時，請重試"}
        except requests.exceptions.ConnectionError:
            logging.error(f"{func.__name__}: 連接錯誤")
            return {"error": "網路連接失敗"}
        except Exception as e:
            logging.error(f"{func.__name__}: {str(e)}", exc_info=True)
            return {"error": f"未知錯誤: {str(e)}"}
    return wrapper

# 使用
@handle_errors
def call_model(self, query, model):
    # 模型調用邏輯
    pass
```

**B. 智能重試機制**

```python
from tenacity import (
    retry,
    stop_after_attempt,
    wait_exponential,
    retry_if_exception_type
)

@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=2, max=10),
    retry=retry_if_exception_type(requests.exceptions.RequestException)
)
def call_api_with_retry(url, payload):
    """帶指數退避的重試"""
    response = requests.post(url, json=payload, timeout=10)
    response.raise_for_status()
    return response.json()
```

**預期改善**: 暫時性錯誤自動恢復率提升至 90%

```python
import tempfile
import os

def atomic_write(filepath, data):
    """原子寫入，防止數據損壞"""
    # 先寫入臨時文件
    fd, temp_path = tempfile.mkstemp(
        dir=os.path.dirname(filepath),
        prefix='.tmp_'
    )

    try:
        with os.fdopen(fd, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

        # 原子替換
        os.replace(temp_path, filepath)
    except:
        os.unlink(temp_path)
        raise
```

**B. 數據備份**

```python
def backup_before_write(filepath, data):
    """寫入前先備份"""
    if os.path.exists(filepath):
        backup_path = f"{filepath}.backup"
        shutil.copy2(filepath, backup_path)

    try:
        atomic_write(filepath, data)
    except:
        # 恢復備份
        if os.path.exists(backup_path):
            shutil.copy2(backup_path, filepath)
        raise
```

        for _ in range(pool_size):
