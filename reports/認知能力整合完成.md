## 🚀 使用示例

### 快速測試

```bash
cd 500/llama32-chat
python3 demos/test_cognitive_capabilities.py
```

### 在代碼中使用

```python
from core.autonomous_agent import autonomous_agent

# 1. 反思決策
result = autonomous_agent.reflect_on_decision(
    decision="選擇本地模型處理簡單對話",
    outcome="成功：響應快速準確",
    context={}
)
print(f"質量評分: {result['quality_score']}/10")

# 2. 分析失敗
analysis = autonomous_agent.analyze_failure_with_cognition(
    failure_event={
        "error_message": "Connection timeout",
        "error_type": "timeout_error"
    }
)
print(f"失敗類型: {analysis['failure_type']}")
print(f"防範措施: {analysis['preventive_measures']}")

# 3. 生成假說
hypotheses = autonomous_agent.generate_hypotheses(
    observation="系統響應緩慢",
    context={}
)
print(f"優先假說: {hypotheses['ranked_hypotheses'][0]}")

# 4. 校準信心
confidence = autonomous_agent.calibrate_confidence(
    prediction="優化將提升性能",
    initial_confidence=0.8,
    supporting_factors=["測試通過", "理論支持"],
    uncertainty_factors=["未測試極端情況"]
)
print(f"校準後信心: {confidence['calibrated_confidence']:.2%}")

# 5. 查看統計
stats = autonomous_agent.get_cognitive_usage_statistics()
print(stats)
```
