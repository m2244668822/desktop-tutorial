### 3.1 檔案系統監控

**Python 實現（使用 watchdog）：**

```python
# monitor_file_changes.py
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
檔案系統即時監控
"""

import time
import hashlib
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

BASE_DIR = Path("/Volumes/智能體/城城城程式")

class ProjectChangeHandler(FileSystemEventHandler):
    """專案變更處理器"""

    def __init__(self):
        self.file_hashes = {}
        self.alerts = []

    def compute_hash(self, filepath):
        """計算檔案哈希"""
        try:
            with open(filepath, 'rb') as f:
                return hashlib.md5(f.read()).hexdigest()
        except Exception:
            return None

    def on_modified(self, event):
        """檔案修改事件"""
        if event.is_directory:
            return

        filepath = Path(event.src_path)

        # 忽略暫存檔案
        if filepath.suffix in ['.pyc', '.pyo', '__pycache__', '.git']:
            return

        # 計算新哈希
        new_hash = self.compute_hash(filepath)

        if new_hash and self.file_hashes.get(str(filepath)) != new_hash:
            self.file_hashes[str(filepath)] = new_hash
            alert = f"📝 檔案變更: {filepath.name}"
            self.alerts.append(alert)
            print(alert)

    def on_created(self, event):
        """新檔案事件"""
        if not event.is_directory:
            print(f"➕ 新檔案: {Path(event.src_path).name}")

    def on_deleted(self, event):
        """檔案刪除事件"""
        if not event.is_directory:
            print(f"❌ 檔案刪除: {Path(event.src_path).name}")
            self.file_hashes.pop(str(event.src_path), None)


def start_monitoring(path=BASE_DIR, recursive=True):
    """啟動監控"""
    event_handler = ProjectChangeHandler()
    observer = Observer()
    observer.schedule(event_handler, str(path), recursive=recursive)
    observer.start()

    print(f"開始監控: {path}")
    print("按 Ctrl+C 停止")

    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        observer.stop()

    observer.join()


if __name__ == "__main__":
    start_monitoring()
```

### 3.2 程序運行狀態追蹤

```python
# monitor_process.py
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
程序運行狀態監控
"""

import psutil
import time
from datetime import datetime

def monitor_process(process_name="python"):
    """監控特定程序"""
    print(f"監控程序: {process_name}")
    print("-" * 50)

    while True:
        found = False
        for proc in psutil.process_iter(['pid', 'name', 'cpu_percent', 'memory_percent', 'create_time']):
            try:
                if process_name.lower() in proc.info['name'].lower():
                    found = True
                    print(f"""
時間: {datetime.now().strftime('%H:%M:%S')}
PID: {proc.info['pid']}
名稱: {proc.info['name']}
CPU: {proc.info['cpu_percent']:.1f}%
記憶體: {proc.info['memory_percent']:.1f}%
                    """)
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                pass

        if not found:
            print(f"未找到程序: {process_name}")

        time.sleep(5)

def get_system_resources():
    """獲取系統資源"""
    cpu = psutil.cpu_percent(interval=1)
    mem = psutil.virtual_memory()
    disk = psutil.disk_usage('/')

    print(f"""
系統資源:
CPU: {cpu}%
記憶體: {mem.percent}% ({mem.used / (1024**3):.1f}GB / {mem.total / (1024**3):.1f}GB)
硬碟: {disk.percent}% ({disk.free / (1024**3):.1f}GB 可用)
    """)

if __name__ == "__main__":
    get_system_resources()
    # monitor_process("python3")
```

## 四、畫面延遲問題診斷

### 4.1 可能原因分析

| 可能原因        | 可能性 | 解決方案                |
| --------------- | ------ | ----------------------- |
| Retina 螢幕渲染 | 高     | 降低解析度或關閉 Retina |
| 記憶體不足      | 高     | 關閉不必要的應用        |
| Ollama 資源佔用 | 高     | 使用更小的模型          |
| pywebview 效能  | 中     | 更新或更換框架          |
| 網路延遲        | 低     | 檢查網路連接            |

### 4.2 排查命令

```bash
# 1. 檢查記憶體使用
top -l 1 | head -20

# 2. 檢查 CPU 使用
top -o cpu

# 3. 檢查磁盤 I/O
iostat -w 1 5

# 4. 檢查網路延遲
ping -c 5 127.0.0.1

# 5. 檢查 GPU 使用
gpuswitcher list 2>/dev/null || echo "無法獲取 GPU 資訊"
```

### 4.3 解決方案
